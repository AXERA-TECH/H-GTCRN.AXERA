cmake_minimum_required(VERSION 3.15)
project(h_gtcrn_core_sdk LANGUAGES CXX C)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include_directories(include)
add_library(h_gtcrn_core_sdk STATIC src/model_runner.cpp)
target_include_directories(h_gtcrn_core_sdk PUBLIC include)
add_executable(h_gtcrn_ax650 examples/main.cpp)
if(BSP_MSP_DIR)
  # Cross compile: -DBSP_MSP_DIR=<ax650n_bsp_sdk>/msp/out
  target_include_directories(h_gtcrn_core_sdk PUBLIC ${BSP_MSP_DIR}/include)
  target_link_directories(h_gtcrn_core_sdk PUBLIC ${BSP_MSP_DIR}/lib)
  target_link_libraries(h_gtcrn_core_sdk PUBLIC ax_engine ax_sys ax_interpreter)
elseif(AX_RUNTIME_ROOT)
  # Native board build: headers/libraries from the device runtime (/soc)
  target_include_directories(h_gtcrn_core_sdk PUBLIC ${AX_RUNTIME_ROOT}/include)
  target_link_directories(h_gtcrn_core_sdk PUBLIC ${AX_RUNTIME_ROOT}/lib)
  target_link_libraries(h_gtcrn_core_sdk PUBLIC ax_engine ax_sys ax_interpreter)
endif()
target_link_libraries(h_gtcrn_ax650 PRIVATE h_gtcrn_core_sdk)
